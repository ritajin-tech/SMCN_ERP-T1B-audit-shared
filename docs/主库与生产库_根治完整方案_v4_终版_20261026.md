### A0 【新·最优先】给清理工具补统一护栏

**为什么必须先做**：`tsPurgeRows` / `tsPurgeSoftLeaves` 走"`getDataRange()` 全表读 → 保留行 `setValues` 整表写回 → `deleteRows` 删尾"。这个模式本身是安全的（源码 9/12 事故后改进），但它**不受 period-lock 管辖**，也没有锁。**用它清理 10 月，等于同时获得了改写 4–9 月的权限** —— 风险不可接受。

另外，**`tsDupReport` 不能用于本次清理**：它的重复定义是"录入时间跨度 ≤ 2 小时"（第 726 行 `windowMs`），而我们 1778 组的跨度是**跨天跨月**。用它 `fix=1` 会**漏掉绝大多数**，且可能误标。

**在 `PeriodLockGuard.js` 末尾追加：**

```js
// ============================================================
// 5. [A0] 清理类工具统一护栏（V4.3.87）
// ------------------------------------------------------------
// 背景：tsPurgeRows / tsPurgeSoftLeaves / tsDupReport(fix=1) / tsTrimBlank /
//       _tsSetValuesGuarded 全部「无锁 + 无期间门禁」，而 purge 类走
//       「全表读 → setValues 写回 → deleteRows 删尾」，一旦调用即可改写或
//       物理删除已冻结期间（B 列 < cutoff）的数据 —— 与 9/12 事故同型。
// 契约：
//   ① 全程持锁（_plAcquire，fail-closed）
//   ② 默认拒绝触碰已关账期间的行；仅当显式 allowClosed=1 才放行，且必须
//      同时给出 allowRows 白名单（行号集合），白名单外的关账行一律拒
//   ③ 目标行 B 列不可解析 → 一律拒（不猜）
//   ④ dry 默认 1：不显式 dry=0 就只报不写
// ============================================================

/**
 * 清理动作的准入检查。**必须在真正的写/删之前调用**，且调用方须在其
 * try 内持锁执行后续动作（本函数返回后锁已释放，故设计为「先准入、
 * 再由调用方在自己的锁内执行」两段式，避免 exec 里做长耗时操作占锁过久）。
 *
 * @param {Object} o
 *   o.sheet        Sheet 对象（必填）
 *   o.targetRows   number[] 目标行号（1-based，必填）
 *   o.allowClosed  bool 是否允许触碰关账期行（默认 false）
 *   o.allowRows    number[] allowClosed=true 时的行号白名单（必填）
 * @returns {{ok:boolean, reason:string, offenders:Array, offenderCount:number, cutoff:string}}
 */
function _plGuardCleanup(o) {
  o = o || {};
  var policy = _plPolicy();
  var cutoff = (policy && policy.cutoff) || PL_CUTOFF;
  var allowClosed = !!o.allowClosed;
  var allowRows = o.allowRows || [];
  var targets = o.targetRows || [];
  var offenders = [];

  var data = o.sheet.getDataRange().getValues();
  var lastRow = data.length;

  for (var i = 0; i < targets.length; i++) {
    var rn = Number(targets[i]);
    if (!rn || rn < 2) { offenders.push({ row: rn, reason: 'PL_REJECT_HEADER_ROW' }); continue; }
    if (rn > lastRow)  { offenders.push({ row: rn, reason: 'PL_REJECT_ROW_GONE' });  continue; }
    var biz = _plParseBizDate(data[rn - 1][1]);
    if (!biz) { offenders.push({ row: rn, reason: 'PL_REJECT_ORIG_UNKNOWN' }); continue; }
    if (!_plIsClosedBizDate(biz, cutoff)) continue;           // 未关账 → 放行
    if (allowClosed && allowRows.indexOf(rn) >= 0) continue;  // 关账但在白名单 → 放行
    offenders.push({ row: rn, reason: 'PL_REJECT_PERIOD_CLOSED' });
  }

  if (offenders.length) {
    return { ok: false, reason: 'PL_REJECT_CLEANUP_OFFENDERS', cutoff: cutoff,
             offenders: offenders.slice(0, 200), offenderCount: offenders.length };
  }
  return { ok: true, reason: '', cutoff: cutoff, offenderCount: 0 };
}

/**
 * 带锁地执行清理动作（purge / 软删 / trim 统一走这里）。
 * @param {Object} o
 *   o.sheet / o.targetRows / o.allowClosed / o.allowRows  同 _plGuardCleanup
 *   o.dry       bool 默认 true —— 不显式 dry=0 就只做准入检查，不执行
 *   o.exec      function(data) 真正的写/删；仅在准入通过且 dry=false 时调用
 *   o.timeoutMs number 可选
 */
function _plGuardedCleanup(o) {
  o = o || {};
  var dry = (o.dry === false) ? false : true;
  var lock = _plAcquire(o.timeoutMs);            // fail-closed：拿不到锁直接抛
  try {
    var chk = _plGuardCleanup(o);
    if (!chk.ok) return chk;
    if (dry) return { ok: true, dry: true, reason: '', cutoff: chk.cutoff };
    var data = o.sheet.getDataRange().getValues();
    var r = o.exec(data);
    try { SpreadsheetApp.flush(); } catch (fe) { /* 与 _plGuardedInsert 同策略，由调用方登记 */ }
    return { ok: true, dry: false, result: r, cutoff: chk.cutoff };
  } finally {
    if (lock) { try { lock.releaseLock(); } catch (re) {} }
  }
}

function tsPurgeRows(e) {
  e = e || {};
  var p = e.parameter || {};
  var body = (e.postBody && typeof e.postBody === 'object') ? e.postBody : {};
  var dry = (String(p.dry || body.dry || '').toLowerCase() === '0') ? false : true; // [A0] 默认 dry

      if (delSoft && isSoft) targetRows.push(i + 1);

    if (!targetRows.length) {
      return JSON.stringify({ ok: true, dry: dry, action: 'purgeRows', removed: 0, msg: '无匹配行' });
    }

    var g = _plGuardedCleanup({
      sheet: sh, targetRows: targetRows, allowClosed: allowClosed, allowRows: allowRows, dry: dry,
      exec: function () {
        var retained = [], removed = 0;
        for (var k = 0; k < data.length; k++) {
          if (k === 0) { retained.push(data[k]); continue; }
          if (targetRows.indexOf(k + 1) >= 0) { removed++; continue; }
          retained.push(data[k]);
        }
        if (retained.length > 1) _tsSetValuesGuarded(sh, retained, data);
        var last = sh.getLastRow();
        if (last > retained.length) sh.deleteRows(retained.length + 1, last - retained.length);
        return { removed: removed, kept: retained.length };
      }
    });

    if (!g.ok) {
      return JSON.stringify({ ok: false, action: 'purgeRows', error: g.reason,
                              offenderCount: g.offenderCount, offenders: g.offenders,
                              hint: '触碰关账期行且未在 allowRows 白名单内；如需放行请显式 allowClosed=1 并附 allowRows' });

                            kept: (g.result && g.result.kept) || 0, cutoff: g.cutoff });
  } catch (err) {
    return JSON.stringify({ ok: false, error: String((err && err.message) || err) });
  }
}

  for (var j = 0; j < rows.length; j++) {
    var r = rows[j] || [];
    var uid = String(r[8] || '').trim();
    if (uid && exist[uid]) { skippedDup++; continue; }
    if (!force && _tsIsLeaveRowForGuard(r)) { skippedLeave++; continue; }

    var out = new Array(14);
    for (var k = 0; k < 14; k++) out[k] = (r[k] === undefined || r[k] === null) ? '' : r[k];
    toWrite.push(out);
  }
  return { rows: toWrite, skippedDup: skippedDup, skippedLeave: skippedLeave };
}

function tsAppendRows(e) {
  e = e || {};
  var p = e.parameter || {};
  var body = e.postBody || {};
  var rows = body.rows || [];
  var force = String(p.force || body.force || '0') === '1';
  var dry = String(p.dry || body.dry || '0') === '1';
  var sheetName = String(p.sheet || body.sheet || 'DB_工时流水');

  if (!rows.length) return JSON.stringify({ ok: false, error: 'no rows provided' });
  if (rows.length > TS_APPEND_MAX_ROWS) {
    return JSON.stringify({ ok: false, error: 'too many rows: ' + rows.length + ' > ' + TS_APPEND_MAX_ROWS });
  }

  try {
    var sh = SpreadsheetApp.openById(TS_ID).getSheetByName(sheetName);
    if (!sh) return JSON.stringify({ ok: false, error: 'sheet not found: ' + sheetName });

    // [A1] dry：只读预演，锁外执行，与写路径共用同一判定函数
    if (dry) {
      var dd = _tsDedupeForAppend(sh, rows, force);
      _tsWriteLog('appendRows(dry=1)', 'req=' + rows.length + ' wouldWrite=' + dd.rows.length +
                  ' skipDup=' + dd.skippedDup + ' skipLeave=' + dd.skippedLeave);
      return JSON.stringify({ ok: true, dry: true, requested: rows.length, wouldWrite: dd.rows.length,
                              skippedDup: dd.skippedDup, skippedLeave: dd.skippedLeave });
    }

    // [A1] 关键变更：rows 原样传入，去重由 lockPreWrite 在【持锁状态】下完成
    var skippedDup = 0, skippedLeave = 0;
    var res = _plGuardedInsert({
      rows: rows,
      dateOf: function (r) { return r[1]; },          // B 列 fillDate
      lockPreWrite: function (rowsIn) {               // [A1] 锁内钩子（guard 第 494-498 行调用）
        var d = _tsDedupeForAppend(sh, rowsIn, force);
        skippedDup = d.skippedDup;
        skippedLeave = d.skippedLeave;
        return { rows: d.rows, duplicates: d.skippedDup };
      },
      bizWrite: function (rows2, policy) {
        var startRow = sh.getLastRow() + 1;
        sh.getRange(startRow, 1, rows2.length, 14).setValues(rows2);
        var back = sh.getRange(startRow, 9, rows2.length, 1).getValues();
        var got = 0;
        for (var m = 0; m < back.length; m++) if (String(back[m][0] || '').trim()) got++;
        return { wrote: rows2.length, verified: got, startRow: startRow };
      }
    });

    // [A1] 全重复时 guard 提前返回 {wrote:0, skippedAll:true} —— 字段缺失要兜住
    var wrote   = (res && res.wrote) ? res.wrote : 0;
    var verified = (res && res.verified !== undefined) ? res.verified : 0;
    var startRow = (res && res.startRow !== undefined) ? res.startRow : null;

    _tsWriteLog('appendRows', 'req=' + rows.length + ' wrote=' + wrote + ' verified=' + verified +
                ' skipDup=' + skippedDup + ' skipLeave=' + skippedLeave);
    return JSON.stringify({ ok: true, requested: rows.length, wrote: wrote, verified: verified,
                            skippedDup: skippedDup, skippedLeave: skippedLeave, startRow: startRow });
  } catch (err) {
    _tsWriteLog('appendRows(ERR)', String((err && err.message) || err));
    return JSON.stringify({ ok: false, error: String((err && err.message) || err) });
  }
}
```

判重比较对目标行与 keeper 行取 A–N 共 14 列，逐列做 `norm()`（去空格 trim，空白视为等同），即 **归一化后的逐列相等**，而非原始字节逐字节相等。

这一定义贯穿 `dedupe_timesheet_exact.py`、`ts_softdel.js` 的 CAS 预检、以及 `tsSoftDelGuarded` 服务端锁内校验，三处口径一致。

### 10.2 主库（请假权威源）全程零影响的证明（Tom 第 5 条后半）

**结论**：软标动作只在生产表 `DB_工时流水` 的 O 列（索引 14）写 `TRUE`，**从不向主库写任何值**；且因为主库底稿的同步链路在结构上已把这些重复行排除在外，软标后主库聚合结果**必然不变**。

**证据链（基于 `masterAttendanceLive.js` 实际源码）**：

1. 主库考勤链的数据流是 **生产表 → 主库底稿 → 按月考勤/月度视图**，且后两步**只从主库底稿 `工时_候选0824` 读取**（`_malBuildAttendData`，行 166–218），**不读生产表**。因此生产表 O 列的变化，经由底稿这一隔离层，不会直接作用于考勤视图。
2. 生产表 → 主库底稿的增量同步 `syncProdToMaster`（行 82–158）对生产表读取做三道排除：
   - 行 114：`_malIsSoftDel(p)` → **O 列=TRUE 的行直接剔除**（即本次软标的目标行，绝不会被同步进底稿）；
   - 行 119：`(pu && baseU[pu]) || baseC[ck]` → 按 **UUID + 内容键去重**（重复行与 keeper 同 UUID，天然跳过）；
   - 行 122：`watermark !== null && pts <= watermark` → **录入时间 ≤ 水位线的行不新增**（本轮 1567 行录入时间全部 ≤ 2026-08-20，远低于当前水位线 ~2026-10-25）。
   三重保险下，这 1567 行在软标前**就已经**被排除在底稿之外——这也正是"主库底稿零重复（distinctUuid 2675 / multiRows 0）"的事实来源。软标只是把 O 列置 TRUE，对上述三道排除逻辑**无任何增量影响**。
3. 因此软标后：底稿新增集合不变 → 按月考勤/月度视图不变。主库全程零影响是**结构性必然**，而非"碰巧没动"。

**可复算的验收步骤（给 Tom 独立核对）**：
- 前置：`syncProdToMaster({dry:true})` 记录 `addCount / addHours / orphanCount`；`rebuildAttendanceLive({dry:true})`、`refreshMasterViews({dry:true})` 记录汇总行数。
- 软标后（GO 阶段）：重跑上述三个 dry 调用，**addCount / addHours / orphanCount 与软标前逐项相等**（因为软标行本就被三道排除挡住，O 翻转不改变它们是否入底稿）；视图 dry 输出的汇总行数同样不变。
- 因底稿是视图的唯一输入、且 O 翻转不增删底稿行，视图必然不变——此步为结构性确认，无需逐人比对。

> 注：若未来有人禁用上述任一道排除（例如改 `syncProdToMaster` 去掉 O 列剔除），则零影响结论不再成立；本方案不改动 `syncProdToMaster` 的排除逻辑（仅 A7 增加"挂起行可见"，不改写入选）。
