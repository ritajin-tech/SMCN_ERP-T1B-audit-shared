## 1. 根因（线上实证）

`_tmp/live_pull/RemoteActions.js`（实拉）：

```
:96-105   var last = sh.getLastRow(); var exist = {};   ← 锁外读 UUID 集合
:107-117  据 exist{} 决策 toWrite（skippedDup / skippedLeave / 同批去重）
:127-138  _plGuardedInsert({ rows: toWrite, dateOf, bizWrite })   ← 未传 lockPreWrite
```

`_plGuardedInsert` 在 `:490` 才取锁。故 `:96-117` 的 `exist{}` 是**取锁前的快照**；
两个并发请求各自读到不含对方在途行的旧快照 → 双方都判定"不重复" → 双写。
这正是 2026-11-01 存量 1,567 组冗余 UUID 的成因。

## 2. 改法（单文件，纯移动去重位置，语义不变）

### 2.1 新增两个纯函数（建议插在 `RemoteActions.js:76` 之后，即 `_tsWriteLog` 之后）

```js
// [Edit A] 读取当前表内 UUID 集合（I 列，1-based 第 9 列）。仅在持锁上下文调用方为权威。
function _tsReadUuidSet(sh) {
  var last = sh.getLastRow();
  var exist = {};
  if (last > 1) {
    var uuids = sh.getRange(2, 9, last - 1, 1).getValues();
    for (var i = 0; i < uuids.length; i++) {
      var u = String(uuids[i][0] || '').trim();
      if (u) exist[u] = true;
    }
  }
  return exist;
}

// [Edit A] 去重 + 造行。纯函数：依赖注入 exist，故锁外/锁内可复用同一逻辑，杜绝双份漂移。
function _tsDedup(rows, exist, force) {
  var out = [], duplicates = 0, leaves = 0;
  for (var j = 0; j < rows.length; j++) {
    var r = rows[j] || [];
    var uid = String(r[8] || '').trim();
    if (uid && exist[uid]) { duplicates++; continue; }
    if (!force && _tsIsLeaveRowForGuard(r)) { leaves++; continue; }
    if (uid) exist[uid] = true;               // 同批内也去重
    var o = new Array(14);
    for (var k = 0; k < 14; k++) o[k] = (r[k] === undefined || r[k] === null) ? '' : r[k];
    out.push(o);
  }
  return { rows: out, duplicates: duplicates, leaves: leaves };
}
```

### 2.2 替换 `RemoteActions.js:96-117`（锁外块）

**原文**（96-117）删除，替换为：

```js
    // [Edit A] 锁外仅作「提示性」预检计数，不参与写入决策。
    // 根因：锁外 exist{} 读到的是取锁前快照，不含并发在途行 → 双写。
    // 权威去重统一在下方 _plGuardedInsert 的 lockPreWrite（:494 锁内）执行。
    var probe = _tsDedup(rows, _tsReadUuidSet(sh), force);
    var toWrite = probe.rows, skippedDup = probe.duplicates, skippedLeave = probe.leaves;
```

### 2.3 修改 `RemoteActions.js:127-138`（`_plGuardedInsert` 调用）

**原文**（127-138）改为：

```js
    var res = _plGuardedInsert({
      rows: toWrite,
      dateOf: function (r) { return r[1]; },        // B 列 fillDate
      // [Edit A] 锁内权威重读 + 二次去重：杜绝并发双写（契约见 PeriodLockGuard.js:494-498）
      lockPreWrite: function (rowsIn, policy) {
        return _tsDedup(rowsIn, _tsReadUuidSet(sh), force);
      },
      bizWrite: function (rows2, policy) {
        var startRow = sh.getLastRow() + 1;
        sh.getRange(startRow, 1, rows2.length, 14).setValues(rows2);
        var back = sh.getRange(startRow, 9, rows2.length, 1).getValues();
        var got = 0;
        for (var m = 0; m < back.length; m++) if (String(back[m][0] || '').trim()) got++;
        return { wrote: rows2.length, verified: got, startRow: startRow };
      }
    });
```
